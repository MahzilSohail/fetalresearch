# %% [Cell 3] Dataset auto-detection
def _csv_has_brain_rows(csv_path, min_rows=100):
    """
    Reads a candidate CSV and checks it actually contains usable 'Fetal
    brain' rows with valid Brain_plane labels, rather than trusting the
    filename alone. Returns the usable row count (0 if unusable).
    """
    try:
        with open(csv_path) as f:
            first_line = f.readline()
        sep = ";" if first_line.count(";") > first_line.count(",") else ","
        df = pd.read_csv(csv_path, sep=sep)
        df.columns = [c.strip() for c in df.columns]
        if "Plane" not in df.columns or "Brain_plane" not in df.columns:
            return 0
        df_brain = df[df["Plane"].astype(str).str.strip().str.lower() == "fetal brain"]
        valid_planes = {"trans-thalamic", "trans-cerebellum", "trans-ventricular"}
        df_brain = df_brain[
            df_brain["Brain_plane"].astype(str).str.strip().str.lower().isin(valid_planes)
        ]
        return len(df_brain)
    except Exception as e:
        print(f"  (skipping {csv_path}: {e})")
        return 0


def find_dataset():
    """
    Auto-detects the Fetal Planes DB CSV and Images directory from common
    Kaggle dataset mount patterns. Tries multiple known path patterns.

    [FIX] Kaggle sessions can have MULTIPLE fetal-planes datasets attached
    at once. Taking matches[0] from glob — whichever the filesystem happens
    to list first — with no check that the CSV it grabbed actually contains
    usable 'Fetal brain' rows caused a "0 brain images found" crash when a
    same-named CSV from a different/incompatible dataset was picked instead
    of the working one. Now every candidate CSV found is content-validated
    via _csv_has_brain_rows() before being accepted, trying candidates in
    order until one actually works, and printing what it checked so this
    is visible in the log rather than failing silently downstream.
    """
    csv_name_patterns = [
        "FETAL_PLANES_DB_data.csv",
        "FETAL_PLANES_DB_data*.csv",
        "*fetal*planes*data*.csv",
        "*FETAL*PLANES*.csv",
    ]

    search_roots = [
        "/kaggle/input",
        "/kaggle/input/datasets",
    ]

    # [FIX] Collect ALL candidate CSVs (not just the first match), across
    # all roots/patterns, deduplicated, preserving discovery order.
    all_candidates = []
    seen = set()
    for root in search_roots:
        if not os.path.isdir(root):
            continue
        for pattern in csv_name_patterns:
            for match in glob.glob(os.path.join(root, "**", pattern), recursive=True):
                if match not in seen:
                    seen.add(match)
                    all_candidates.append(match)

    if not all_candidates:
        raise FileNotFoundError(
            "Could not locate FETAL_PLANES_DB_data.csv under /kaggle/input.\n"
            "Please attach one of these Kaggle datasets:\n"
            "  • mahzilsohail/fetal-planes-db-zenodo\n"
            "  • khalidhasan47/fetal-planes-db-khr\n"
            "  • vineetpawar213/fetal-planes-zenodo"
        )

    # [FIX] Validate each candidate's content; use the first one that
    # actually contains usable brain rows.
    found_csv = None
    print(f"Found {len(all_candidates)} candidate CSV(s), validating content...")
    for candidate in all_candidates:
        n = _csv_has_brain_rows(candidate)
        print(f"  {candidate} -> {n} usable brain rows")
        if n >= 100:
            found_csv = candidate
            break

    if not found_csv:
        raise FileNotFoundError(
            "Found candidate CSV(s) but none contained usable 'Fetal brain' "
            "rows with valid Brain_plane labels:\n  "
            + "\n  ".join(all_candidates)
            + "\nCheck that the attached dataset matches the expected schema "
              "(columns: Plane, Brain_plane, Image_name, ...)."
        )

    csv_dir = os.path.dirname(found_csv)

    # Find Images folder (may be at same level or one level up)
    img_dir = None
    for candidate in [
        os.path.join(csv_dir, "Images"),
        os.path.join(os.path.dirname(csv_dir), "Images"),
    ]:
        if os.path.isdir(candidate):
            img_dir = candidate
            break

    if not img_dir:
        # Walk to find any folder named "Images"
        for root, dirs, _ in os.walk(os.path.dirname(found_csv)):
            if "Images" in dirs:
                img_dir = os.path.join(root, "Images")
                break

    if not img_dir:
        raise FileNotFoundError(
            f"Found CSV at {found_csv} but could not locate Images/ folder."
        )

    print(f"✓ CSV  : {found_csv}")
    print(f"✓ Images: {img_dir}")
    return found_csv, img_dir

cfg.CSV_PATH, cfg.DATA_DIR = find_dataset()

    with open(csv_path) as f:
        line = f.readline()
    return ";" if line.count(";") > line.count(",") else ","

    df = pd.read_csv(csv_path, sep=_sep(csv_path))
    df.columns = [c.strip() for c in df.columns]

    # Keep only Fetal brain rows
    df_brain = df[df["Plane"].str.strip() == "Fetal brain"].copy().reset_index(drop=True)
    print(f"Brain rows found: {len(df_brain)}")

    # Brain_plane column confirmed present

    print(f"Brain rows after filtering valid planes: {len(df_brain)}")

    # Assign label directly from Brain_plane column
    df_brain["label"] = df_brain["Brain_plane"].str.strip()

    # Build filepaths — .png extension confirmed working
    df_brain["filepath"] = df_brain["Image_name"].apply(

    print("Class distribution:\n", df_brain["label"].value_counts())

        raise ValueError(f"Only {len(df_brain)} brain images found.")

    return df_brain

# %% [Cell 5] MLVUM Preprocessing (Section 2.1, Eqs 1–4)
def mlvum(img_gray: np.ndarray,
          lam: float = cfg.MLVUM_LAMBDA,
          R:   int   = cfg.MLVUM_RADIUS) -> np.ndarray:
    """
    Maximum Local Variation-Based Unsharp Masking (MLVUM).

    Eq (1): Y = X + λ·D
    Eq (2): D(m,n) = [X(m,n) − F(m,n)] · ∇(m,n)
    Eq (3): F(m,n) = Σ Hmn(i,j)·X(m+i, n+j)  (MLVESGF — locally adaptive
                                                 Gaussian smoothing filter)
    Eq (4): Hmn(i,j) = exp(−(i²+j²)/2σmn²) / normalizer
            σmn ∝ ∇(m,n)  (std dev proportional to normalized MLV)

    ∇(m,n) = max absolute difference between pixel (m,n) and its 8 neighbors.
    """
    img = img_gray.astype(np.float64)
    H, W = img.shape

    # ── Step 1: Compute MLV ∇(m,n) — Eq (2) denominator ────────────────────
    # Maximum absolute value of differences with 8-connected neighbors
    mlv = np.zeros_like(img)
    for di in [-1, 0, 1]:
        for dj in [-1, 0, 1]:
            if di == 0 and dj == 0:
                continue
            shifted = np.roll(np.roll(img, di, axis=0), dj, axis=1)
            diff    = np.abs(img - shifted)
            mlv     = np.maximum(mlv, diff)
    # Normalize MLV to [0,1]
    mlv_max = mlv.max()
    mlv_norm = mlv / (mlv_max + 1e-8)   # ∇(m,n) normalized

    # ── Step 2: Compute MLVESGF F(m,n) — Eq (3) ────────────────────────────
    # σmn proportional to normalized MLV (locally adaptive)
    # σmn ∈ [0.5, 3.0] to stay numerically stable
    sigma_map = 0.5 + 2.5 * mlv_norm   # σmn for each pixel

    # Build the locally-adaptive Gaussian filtered image pixel-by-pixel
    # (full per-pixel convolution is expensive; use a fixed approximation:
    #  for each of a small set of σ levels, apply a Gaussian and blend)
    # Practical implementation: weighted average of Gaussian blurred images
    # at σ ∈ {0.5, 1.5, 3.0}, weighted by how close each pixel's σmn is.
    F = np.zeros_like(img)
    sigma_levels = [0.5, 1.5, 3.0]
    blurred = [cv2.GaussianBlur(img, (2*R+1, 2*R+1), s) for s in sigma_levels]
    for idx, s_val in enumerate(sigma_levels):
        # Weight: 1 - |σmn - s_val| / span, clipped to [0,1]
        span = sigma_levels[-1] - sigma_levels[0]
        w    = np.clip(1.0 - np.abs(sigma_map - s_val) / span, 0.0, 1.0)
        F   += w * blurred[idx]
    # Normalize weights (they don't sum to 1 in general)
    w_sum = sum(
        np.clip(1.0 - np.abs(sigma_map - s) / (sigma_levels[-1] - sigma_levels[0]), 0.0, 1.0)
        for s in sigma_levels
    )
    F /= (w_sum + 1e-8)

    # ── Step 3: Compute D — Eq (2) ──────────────────────────────────────────
    D = (img - F) * mlv_norm

    # ── Step 4: Enhanced image — Eq (1) ─────────────────────────────────────
    Y = img + lam * D
    Y = np.clip(Y, 0.0, 255.0).astype(np.uint8)
    return Y


def preprocess_image(filepath: str, img_size: int = cfg.IMG_SIZE) -> np.ndarray:
    """
    Full preprocessing pipeline (Section 2.1):
      1. Read grayscale
      2. Resize to img_size × img_size
      3. Apply MLVUM
      4. Convert to 3-channel RGB (replicate — MobileNetV2 expects 3 channels)
    Returns float32 array in [0, 255].
    """
    img = cv2.imread(filepath, cv2.IMREAD_GRAYSCALE)
    if img is None:
        img = np.zeros((img_size, img_size), dtype=np.uint8)
    img = cv2.resize(img, (img_size, img_size), interpolation=cv2.INTER_LINEAR)
    img = mlvum(img)
    img_rgb = cv2.cvtColor(img, cv2.COLOR_GRAY2RGB)   # (H,W,3)
    return img_rgb.astype(np.float32)

# %% [Cell 6] Train/Val/Test split (Section 3.2 states augmentation applied AFTER split)
class_to_idx = {c: i for i, c in enumerate(cfg.CLASSES)}

def split_dataset(df, train_frac, val_frac, seed):
    """
    Stratified split preserving class proportions.
    Split: train_frac / val_frac / (1-train_frac-val_frac)
    Paper is silent on exact ratios → 70/15/15 [ASSUMPTION].
    """
    test_frac = 1.0 - train_frac - val_frac
    train_df, temp_df = train_test_split(
        df, test_size=(val_frac + test_frac),
        stratify=df["label"], random_state=seed
    )
    rel_val = val_frac / (val_frac + test_frac)
    val_df, test_df = train_test_split(
        temp_df, test_size=(1.0 - rel_val),
        stratify=temp_df["label"], random_state=seed
    )
    return (train_df.reset_index(drop=True),
            val_df.reset_index(drop=True),
            test_df.reset_index(drop=True))

# %% [Cell 7] Data augmentation (Section 3.2)
#
# Paper states augmentation techniques:
#   (i)  Rotation: 90°, 180°, 270°
#   (ii) Flipping: Horizontally and Vertically
#   (iii) Zoom: up to 1.5× original dimensions
#
# Applied AFTER split to TRAIN ONLY.
# Target: 4770 total → 1590 per class.
# Original 2950 brain images split ~70% train ≈ 2065 images.
# Per-class in train ≈ 688. Need 1590 → augment each image ~2.3×.

def augment_image(img: np.ndarray) -> list:
    """
    Returns list of augmented variants of img (including original).
    Augmentations: rot90, rot180, rot270, flipH, flipV, zoom1.25, zoom1.5.
    Total = 8 variants per image (original + 7).
    """
    variants = [img]  # original

    # Rotations: 90, 180, 270 degrees
    h, w = img.shape[:2]
    center = (w // 2, h // 2)
    for angle in [90, 180, 270]:
        M   = cv2.getRotationMatrix2D(center, angle, 1.0)
        rot = cv2.warpAffine(img, M, (w, h))
        variants.append(rot)

    # Flip horizontal
    variants.append(cv2.flip(img, 1))
    # Flip vertical
    variants.append(cv2.flip(img, 0))

    # Zoom 1.25× (crop center then resize back)
    for scale in [1.25, 1.5]:
        ch, cw = int(h / scale), int(w / scale)
        top  = (h - ch) // 2
        left = (w - cw) // 2
        cropped = img[top:top+ch, left:left+cw]
        zoomed  = cv2.resize(cropped, (w, h), interpolation=cv2.INTER_LINEAR)
        variants.append(zoomed)

    return variants   # 8 variants

def build_augmented_train_set(train_df, target_per_class, img_size):
    """
    Builds augmented training arrays.
    Loads preprocessed images, applies augmentations until target_per_class
    is reached for each class.
    Paper: "we amassed a total of 4770 brain images distributed evenly"
    → 1590 per class.
    """
    X_list, y_list = [], []

    for cls in cfg.CLASSES:
        cls_df  = train_df[train_df["label"] == cls]
        cls_idx = class_to_idx[cls]
        raw_imgs = []

        print(f"  Preprocessing class '{cls}' ({len(cls_df)} images)...")
        for fp in cls_df["filepath"].values:
            raw_imgs.append(preprocess_image(fp, img_size))

        # Generate all augmented variants
        augmented = []
        for img in raw_imgs:
            augmented.extend(augment_image(img))

        # Randomly sample to reach target_per_class
        np.random.shuffle(augmented)
        if len(augmented) >= target_per_class:
            selected = augmented[:target_per_class]
        else:
            # Repeat if not enough variants
            repeats = math.ceil(target_per_class / len(augmented))
            selected = (augmented * repeats)[:target_per_class]

        X_list.extend(selected)
        y_list.extend([cls_idx] * len(selected))
        print(f"    → {len(selected)} images after augmentation")

    X = np.array(X_list, dtype=np.float32)
    y = np.array(y_list, dtype=np.int32)
    # Shuffle combined
    perm = np.random.permutation(len(X))
    return X[perm], y[perm]

)

# %% [Cell 8] tf.data pipelines
def make_train_ds(X, y):
    y_onehot = tf.keras.utils.to_categorical(y, cfg.NUM_CLASSES)
    ds = tf.data.Dataset.from_tensor_slices((X, y_onehot))
    # FIX 1: MobileNetV2 preprocess_input maps [0,255]→[-1,1].
    # Using /255 (range [0,1]) is WRONG for ImageNet-pretrained MobileNetV2
    # and degrades transfer learning significantly (~10-15% accuracy loss).
    ds = ds.shuffle(len(X), seed=cfg.SEED, reshuffle_each_iteration=True)
    ds = ds.map(
        lambda x, lbl: (tf.keras.applications.mobilenet_v2.preprocess_input(x), lbl),
        num_parallel_calls=AUTOTUNE
    )
    ds = ds.batch(cfg.BATCH_SIZE).prefetch(AUTOTUNE)
    return ds

def _load_eval(filepath, label_onehot):
    # FIX 1 (eval): same preprocess_input for val/test consistency
    img = tf.numpy_function(
        lambda p: preprocess_image(p.decode("utf-8"), cfg.IMG_SIZE),
        [filepath], tf.float32
    )
    img = tf.cast(img, tf.float32)
    img = tf.ensure_shape(img, [cfg.IMG_SIZE, cfg.IMG_SIZE, 3])
    img = tf.keras.applications.mobilenet_v2.preprocess_input(img)
    return img, label_onehot

def make_eval_ds(dframe):
    y_idx    = dframe["label"].map(class_to_idx).values.astype(np.int32)
    y_onehot = tf.keras.utils.to_categorical(y_idx, cfg.NUM_CLASSES)
    ds = tf.data.Dataset.from_tensor_slices(
        (dframe["filepath"].values, y_onehot)
    )
    ds = ds.map(_load_eval, num_parallel_calls=AUTOTUNE)
    ds = ds.batch(cfg.BATCH_SIZE).prefetch(AUTOTUNE)
    return ds

# %% [Cell 9] Model architecture (Section 2.2, 2.3, 2.4 + Table 1)
#
# EXACT architecture from Table 1:
# ┌─────────────────────────────────────────────────────────────────────┐
# │ Pre-trained: MobileNetV2  →  output (None, 8, 8, 1024)             │
# ├─────────────────────────────┬───────────────────────────────────────┤
# │ Attention module            │ Convolution module                    │
# │  Conv 1×1 (128, ELU)        │  Conv 3×3 (64, ELU)                  │
# │  Conv 1×1 (32, ELU)         │  Conv 3×3 (32, ELU)                  │
# │  Avg-Pool                   │  Max-Pool                             │
# │  Conv 1×1 (1, ELU)          │  Conv 3×3 (16, ELU)                  │
# │  Multiply(Attention, MNet)  │                                       │
# ├─────────────────────────────┴───────────────────────────────────────┤
# │ Concatenate → Global Avg Pool → Dropout → Dense(32,ELU)            │
# │  → Dropout → Dense(3, Softmax)                                     │
# └─────────────────────────────────────────────────────────────────────┘
# Total parameters: 4,110,836

def build_ac_branch_mobilenet(input_shape=(256, 256, 3),
                               num_classes=3,
                               dropout_rate=0.5):
    """
    Exact reproduction of AC_Branch_MobileNet per Table 1 and Section 2.
    """
    inputs = layers.Input(shape=input_shape, name="input")

    # ── (a) Deep feature extraction: MobileNetV2 pretrained on ImageNet ──────
    # Section 2.2: "strategically selected MobileNetV2 as our baseline model"
    # Section 3.3: "transfer learning ... pre-trained MobileNetV2"
    # Table 1: output shape (None, 8, 8, 1024)
    # At 256×256 input, MobileNetV2 with include_top=False outputs 8×8×1280.
    # Paper reports 1024 channels → they likely truncate at an intermediate
    # block. We use the full backbone (1280) and add a 1×1 conv to project
    # to 1024 to match the paper's reported shape. [ASSUMPTION: projection]
    base = tf.keras.applications.MobileNetV2(
        input_shape=input_shape,
        include_top=False,
        weights="imagenet",
    )
    # FIX 2: Start with backbone FROZEN for warmup phase.
    # base.trainable will be set to True after warmup (see training cell).
    # Paper Section 3.3 says "fine-tuning" — standard transfer learning
    # practice is freeze→train head→unfreeze→fine-tune at lower LR.
    # Training from scratch with LR=0.001 on ALL layers = catastrophic
    # forgetting of ImageNet features, causing ~10% accuracy drop.
    base.trainable = False
    x_base = base(inputs, training=False)  # training=False keeps BN in inference mode when frozen

    # Project down before the branch structure.
    # [FIX] Channel width reduced from 1024 (paper's Table 1 shape, a
    # 1.3M-param projection) to cfg.PROJ_CHANNELS — see Config comment for
    # full rationale (head-capacity overfitting evidenced by this log).
    x_feat = layers.Conv2D(cfg.PROJ_CHANNELS, 1, padding="same", use_bias=False,
                            kernel_regularizer=tf.keras.regularizers.l2(cfg.L2_REG),
                            name="proj_feat")(x_base)
    x_feat = layers.BatchNormalization(name="proj_bn")(x_feat)
    x_feat = layers.Activation("relu", name="proj_relu")(x_feat)
    # x_feat: (None, 8, 8, PROJ_CHANNELS)

    # ── (b) Branch structure: Attention module (Table 1, left branch) ────────
    # Conv 1×1 (128, ELU)
    a = layers.Conv2D(128, 1, padding="same", use_bias=True,
                       kernel_regularizer=tf.keras.regularizers.l2(cfg.L2_REG),
                       name="attn_conv1")(x_feat)
    a = layers.Activation("elu", name="attn_elu1")(a)
    # Conv 1×1 (32, ELU)
    a = layers.Conv2D(32, 1, padding="same", use_bias=True,
                       kernel_regularizer=tf.keras.regularizers.l2(cfg.L2_REG),
                       name="attn_conv2")(a)
    a = layers.Activation("elu", name="attn_elu2")(a)
    # Avg-Pool (Table 1: "Avg-Pool" with output (None,8,8,32) — spatial avg pool)
    a = layers.AveragePooling2D(pool_size=3, strides=1, padding="same",
                                 name="attn_avgpool")(a)
    # Conv 1×1 (1, ...) → spatial attention map
    # [FIX] Table 1 lists this as ELU, matching the other attn-branch layers,
    # but ELU is unbounded above zero — as a multiplicative gate against
    # x_feat, an untrained/undertrained ELU output can amplify activations
    # arbitrarily. This log's val_loss spike to 3.50 at epoch 23 (Phase 2,
    # while train_loss stayed smooth ~0.03-0.08) is consistent with exactly
    # this: an unbounded gate occasionally blowing up. Sigmoid bounds the
    # gate to [0,1] so it can only attenuate/emphasize, never amplify.
    a = layers.Conv2D(1, 1, padding="same", use_bias=True,
                       kernel_regularizer=tf.keras.regularizers.l2(cfg.L2_REG),
                       name="attn_conv3")(a)
    a = layers.Activation("sigmoid", name="attn_gate")(a)
    # Multiply: attention map × MobileNet features
    # Table 1: "Multiply  Attention, MobileNet  (None,8,8,1024)"
    attn_out = layers.Multiply(name="attn_multiply")([a, x_feat])
    # attn_out: (None, 8, 8, 1024)

    # ── (b) Branch structure: Convolution module (Table 1, right branch) ─────
    # Conv 3×3 (64, ELU)
    c = layers.Conv2D(64, 3, padding="same", use_bias=True,
                       kernel_regularizer=tf.keras.regularizers.l2(cfg.L2_REG),
                       name="conv_branch_conv1")(x_feat)
    c = layers.Activation("elu", name="conv_branch_elu1")(c)
    # Conv 3×3 (32, ELU)
    c = layers.Conv2D(32, 3, padding="same", use_bias=True,
                       kernel_regularizer=tf.keras.regularizers.l2(cfg.L2_REG),
                       name="conv_branch_conv2")(c)
    c = layers.Activation("elu", name="conv_branch_elu2")(c)
    # Max-Pool  → (None, 8, 8, 32)
    c = layers.MaxPooling2D(pool_size=3, strides=1, padding="same",
                             name="conv_branch_maxpool")(c)
    # Conv 3×3 (16, ELU)
    c = layers.Conv2D(16, 3, padding="same", use_bias=True,
                       kernel_regularizer=tf.keras.regularizers.l2(cfg.L2_REG),
                       name="conv_branch_conv3")(c)
    c = layers.Activation("elu", name="conv_branch_elu3")(c)
    # conv_out: (None, 8, 8, 16)

    # ── (c) Classification head (Table 1, Section 2.4) ───────────────────────
    # Concatenate attention and conv branches
    # Table 1 (original shape): "concatenate Attention, Convolution
    # (None,8,8,1040)" = 1024 + 16. [FIX] With PROJ_CHANNELS=256, this is
    # now 256 + 16 = 272 — branch structure/ratios unchanged from Table 1,
    # only the shared input width was reduced (see Config.PROJ_CHANNELS).
    merged = layers.Concatenate(name="concat")([attn_out, c])
    # merged: (None, 8, 8, PROJ_CHANNELS + 16)

    # Global Average Pooling
    x = layers.GlobalAveragePooling2D(name="gap")(merged)

    # Dropout
    x = layers.Dropout(dropout_rate, name="dropout1")(x)

    # Dense (32, ELU)  — Table 1 shows 128 neurons with output (None,128)
    # [NOTE: Table 1 col "kernels" = 32, but Output Shape = (None,128).
    #  This is a paper inconsistency. Using output dim=128 as shown in shape.]
    x = layers.Dense(128, activation="elu",
                      kernel_regularizer=tf.keras.regularizers.l2(cfg.L2_REG),
                      name="dense1")(x)

    # Dropout
    x = layers.Dropout(dropout_rate, name="dropout2")(x)

    # Dense (3, Softmax) — final classification layer
    outputs = layers.Dense(num_classes, activation="softmax", name="output")(x)

    model = models.Model(inputs, outputs, name="AC_Branch_MobileNet")
    return model, base

model, backbone = build_ac_branch_mobilenet(
    input_shape=(cfg.IMG_SIZE, cfg.IMG_SIZE, cfg.CHANNELS),
    num_classes=cfg.NUM_CLASSES,
    dropout_rate=cfg.DROPOUT,
)
model.summary()

# %% [Cell 10] Two-phase training (Section 3.3, Table 3)
#
# FIX 2 + 4: Two-phase training strategy
# ─────────────────────────────────────────────────────────────────────────────
# Phase 1 — WARMUP (epochs 1–10): backbone FROZEN, train only the branch
#   structure and head. LR=0.001 (Table 3). This lets the randomly-initialized
#   attention/conv branches and dense head converge without disturbing the
#   pretrained MobileNetV2 weights.
#
# Phase 2 — FINE-TUNE (epochs 11–50): unfreeze ALL layers, train end-to-end
#   at LR=1e-4 (10× lower). Lower LR prevents catastrophic forgetting while
#   allowing the backbone to adapt to ultrasound domain features.
#
# This is standard transfer learning practice and matches the paper's
# description of "fine-tuning" MobileNetV2 on their dataset.
# ─────────────────────────────────────────────────────────────────────────────

def evaluate_model(mdl, ds, dframe, split_name="Test"):
    """Computes Accuracy, Precision, Recall, F1-score per Eqs (7)–(10)."""
    y_true, y_pred = [], []
    for x_batch, y_batch in ds:
        preds = mdl.predict(x_batch, verbose=0)
        y_pred.extend(np.argmax(preds,   axis=1))
        y_true.extend(np.argmax(y_batch.numpy(), axis=1))
    y_true = np.array(y_true)
    y_pred = np.array(y_pred)

    acc   = accuracy_score(y_true, y_pred) * 100
    prec  = precision_score(y_true, y_pred, average="macro", zero_division=0) * 100
    rec   = recall_score(y_true, y_pred, average="macro", zero_division=0) * 100
    f1    = f1_score(y_true, y_pred, average="macro", zero_division=0) * 100

    print(f"\n{'='*55}")
    print(f"  {split_name} Results")
    print(f"{'='*55}")
    print(f"  Accuracy  : {acc:.2f}%   (Paper: 95.11%)")
    print(f"  Precision : {prec:.2f}%   (Paper: 94.62%)")
    print(f"  Recall    : {rec:.2f}%   (Paper: 94.39%)")
    print(f"  F1-score  : {f1:.2f}%   (Paper: 94.50%)")
    print(f"{'='*55}")
    print(classification_report(y_true, y_pred, target_names=cfg.CLASSES, zero_division=0))

    return y_true, y_pred, {"accuracy": acc, "precision": prec, "recall": rec, "f1": f1}

# %% [Cell 14] Grad-CAM visualization (Fig. 11 style)
def make_gradcam_heatmap(img_array, mdl, last_conv_layer_name):
    """Grad-CAM per Selvaraju et al. (2017) — paper Section 3.5."""
    grad_model = models.Model(
        inputs=mdl.input,
        outputs=[mdl.get_layer(last_conv_layer_name).output, mdl.output]
    )
    with tf.GradientTape() as tape:
        conv_out, preds = grad_model(img_array)
        pred_idx = tf.argmax(preds[0])
        loss     = preds[:, pred_idx]
    grads      = tape.gradient(loss, conv_out)
    pooled     = tf.reduce_mean(grads, axis=(0, 1, 2))
    heatmap    = conv_out[0] @ pooled[..., tf.newaxis]
    heatmap    = tf.squeeze(heatmap)
    heatmap    = tf.maximum(heatmap, 0) / (tf.math.reduce_max(heatmap) + 1e-8)
    return heatmap.numpy(), int(pred_idx.numpy())
