CONFIG = {

# =============================================================================
# SECTION 2.2.1 — DATA ACQUISITION
# Paper: Fetal Planes dataset (Burgos-Artizzu et al. 2020), 12,400 images in
# 6 classes. Only maternal cervix, brain, and thorax kept (relatively balanced:
# 1718 thorax, 1626 cervix, 3092 brain).
#
# NOTE: the Kaggle mirror "Fetal_Planes_DB_dataset" (minhnhtl05) is already
# pre-split into train/val/test folders, each containing one subfolder per
# class ("Fetal abdomen", "Fetal brain", "Fetal femur", "Fetal thorax",
# "Maternal cervix", "Other"). We use that split directly instead of a CSV.
# =============================================================================

# %% CELL 4 — Locate the dataset root, then build train/val/test dataframes
def find_dataset_root(search_root, folder_name):
    """Walk /kaggle/input looking for a folder named `folder_name` that
    directly contains train/, val/, and test/ subfolders."""
    for root, dirs, _files in os.walk(search_root):
        if folder_name in dirs:
            candidate = os.path.join(root, folder_name)
            if all(os.path.isdir(os.path.join(candidate, s)) for s in ("train", "val", "test")):
                return candidate
    raise FileNotFoundError(
        f"Could not locate a '{folder_name}' folder with train/val/test "
        f"subfolders under {search_root}. Run the directory-listing snippet "
        f"again and update CONFIG['dataset_folder_name'] if the name differs."
    )

DATASET_ROOT = find_dataset_root(CONFIG["dataset_search_root"], CONFIG["dataset_folder_name"])
print("Dataset root found at:", DATASET_ROOT)


def build_split_dataframe(dataset_root, split_name, classes):
    rows = []
    split_dir = os.path.join(dataset_root, split_name)
    for cls in classes:
        cls_dir = os.path.join(split_dir, cls)
        if not os.path.isdir(cls_dir):
            print(f"WARNING: class folder not found, skipping: {cls_dir}")
            continue
        for fname in os.listdir(cls_dir):
            if fname.lower().endswith((".png", ".jpg", ".jpeg")):
                rows.append({"filepath": os.path.join(cls_dir, fname), "Plane": cls})
    return pd.DataFrame(rows)

train_df = build_split_dataframe(DATASET_ROOT, "train", CONFIG["classes"])
val_df   = build_split_dataframe(DATASET_ROOT, "val",   CONFIG["classes"])
test_df  = build_split_dataframe(DATASET_ROOT, "test",  CONFIG["classes"])

if len(train_df) == 0 or len(val_df) == 0 or len(test_df) == 0:
    raise RuntimeError(
        "One of train/val/test came back empty. Print os.listdir(DATASET_ROOT) "
        "and os.listdir(os.path.join(DATASET_ROOT, 'train')) to check the exact "
        "class-folder names and fix CONFIG['classes'] to match exactly."
    )

print(f"Train: {len(train_df)} | Val: {len(val_df)} | Test: {len(test_df)}")
for name, part in [("train", train_df), ("val", val_df), ("test", test_df)]:
    print(name, part["Plane"].value_counts().to_dict())

# =============================================================================
# SECTION 2.2.2/2.2.3 — ML METHODOLOGY & TRAINING
# Object classification (not detection) chosen for MLOps-friendliness.
# Three architectures: VGG19, ResNet50, MobileNetV2, ImageNet-initialized,
# with a custom head: Conv(512,3x3)+BN+ReLU -> Conv(3,1x1)+BN -> GAP -> BN
# -> Softmax. SGD, lr=0.001, batch=32, 100 epochs, categorical cross-entropy.
# Preprocessing: resize to 224x224x3, mean subtraction, on-the-fly augmentation
# (rotation +-30deg, brightness jitter, horizontal flip).
# =============================================================================

# %% CELL 6 — Data generators (mean subtraction + augmentation, per paper)
IMG_SIZE = CONFIG["img_size"]
BATCH = CONFIG["batch_size"]

train_datagen = ImageDataGenerator(
    featurewise_center=True,          # "removing the intensity average"
    rotation_range=30,                # random rotation -30..+30 deg
    brightness_range=[0.7, 1.3],      # random brightness correction
    horizontal_flip=True,             # random flipping
)
eval_datagen = ImageDataGenerator(featurewise_center=True)

# Estimate per-channel mean on a sample of training images for featurewise_center
sample_paths = train_df["filepath"].sample(min(500, len(train_df)), random_state=42)
sample_imgs = []
for p in sample_paths:
    img = cv2.imread(p)
    if img is None:
        continue
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img = cv2.resize(img, (IMG_SIZE, IMG_SIZE))
    sample_imgs.append(img)
sample_imgs = np.array(sample_imgs, dtype=np.float32)
train_datagen.fit(sample_imgs)
eval_datagen.mean = train_datagen.mean
eval_datagen.std = train_datagen.std

train_gen = train_datagen.flow_from_dataframe(
    train_df, x_col="filepath", y_col="Plane",
    target_size=(IMG_SIZE, IMG_SIZE), batch_size=BATCH,
    class_mode="categorical", classes=CONFIG["classes"], shuffle=True, seed=42,
)
val_gen = eval_datagen.flow_from_dataframe(
    val_df, x_col="filepath", y_col="Plane",
    target_size=(IMG_SIZE, IMG_SIZE), batch_size=BATCH,
    class_mode="categorical", classes=CONFIG["classes"], shuffle=False,
)
test_gen = eval_datagen.flow_from_dataframe(
    test_df, x_col="filepath", y_col="Plane",
    target_size=(IMG_SIZE, IMG_SIZE), batch_size=BATCH,
    class_mode="categorical", classes=CONFIG["classes"], shuffle=False,
)

# %% CELL 7 — Model builder (shared custom head across all 3 backbones)
def build_model(backbone_name: str, num_classes: int = 3, input_shape=(224, 224, 3)):
    if backbone_name == "VGG19":
        base = VGG19(weights="imagenet", include_top=False, input_shape=input_shape)
    elif backbone_name == "ResNet50":
        base = ResNet50(weights="imagenet", include_top=False, input_shape=input_shape)
    elif backbone_name == "MobileNetV2":
        base = MobileNetV2(weights="imagenet", include_top=False, input_shape=input_shape)
    else:
        raise ValueError(backbone_name)

    base.trainable = True  # fine-tune end-to-end (transfer learning init from ImageNet)

    x = base.output
    # First conv layer: 512 filters, 3x3, stride 1, same padding + BN + ReLU
    x = layers.Conv2D(512, (3, 3), strides=1, padding="same", name="head_conv1")(x)
    x = layers.BatchNormalization(name="head_bn1")(x)
    x = layers.ReLU(name="head_relu1")(x)
    # Second conv layer: 3 filters (= num_classes), 1x1, stride 1, same padding + BN
    x = layers.Conv2D(num_classes, (1, 1), strides=1, padding="same", name="head_conv2")(x)
    x = layers.BatchNormalization(name="head_bn2")(x)
    # Global average pooling to reduce params/overfitting, then BN
    x = layers.GlobalAveragePooling2D(name="head_gap")(x)
    x = layers.BatchNormalization(name="head_bn3")(x)
    outputs = layers.Activation("softmax", name="predictions")(x)

    model = models.Model(inputs=base.input, outputs=outputs, name=f"{backbone_name}_fetal")
    return model

# %% CELL 8 — Train all three models with CodeCarbon tracking (Section 2.4.2)
architectures = ["VGG19", "ResNet50", "MobileNetV2"]
histories = {}
trained_models = {}
emissions_summary = []

for arch in architectures:
    print(f"\n{'='*60}\nTraining {arch}\n{'='*60}")

    model = build_model(arch, num_classes=len(CONFIG["classes"]))
    model.compile(
        optimizer=optimizers.SGD(learning_rate=CONFIG["learning_rate"]),
        loss="categorical_crossentropy",
        metrics=["accuracy"],
    )

    tracker = EmissionsTracker(
        project_name=f"FetalMLOps_{arch}",
        output_dir=CONFIG["output_dir"],
        output_file=f"emissions_{arch}.csv",
        log_level="error",
    )
    tracker.start()

    history = model.fit(
        train_gen,
        validation_data=val_gen,
        epochs=CONFIG["epochs"],
        verbose=1,
    )

    co2_kg = tracker.stop()
    emissions_summary.append({"model": arch, "kg_CO2": co2_kg})

    histories[arch] = history.history
    trained_models[arch] = model
    model.save(os.path.join(CONFIG["output_dir"], f"{arch}_fetal.keras"))

emissions_df = pd.DataFrame(emissions_summary)
print("\nEstimated carbon emissions (screening phase, 100 epochs each):")
print(emissions_df)
emissions_df.to_csv(os.path.join(CONFIG["output_dir"], "emissions_summary.csv"), index=False)

    axes[0, 1].plot(h["loss"], label=arch)
    axes[1, 0].plot(h["val_accuracy"], label=arch)

axes[0, 0].set_title("Training Accuracy per Epoch")
axes[0, 1].set_title("Training Loss per Epoch")
axes[1, 0].set_title("Validation Accuracy per Epoch")
axes[1, 1].set_title("Validation Loss per Epoch")
for ax in axes.flat:
    ax.set_xlabel("Epoch")

# %% CELL 10 — Precision / Recall / F1 per class, per model (Table 1)
results_table = []
predictions_store = {}

for arch in architectures:
    model = trained_models[arch]
    test_gen.reset()
    y_pred_probs = model.predict(test_gen, verbose=0)
    y_true = test_gen.classes
    y_pred = np.argmax(y_pred_probs, axis=1)
    predictions_store[arch] = (y_true, y_pred, y_pred_probs)

    precision, recall, f1, _ = precision_recall_fscore_support(
        y_true, y_pred, labels=range(len(CONFIG["classes"]))
    )
    for i, cls in enumerate(CONFIG["classes"]):
        results_table.append({
            "Model": arch, "Class": cls,
            "Precision": round(precision[i], 3),
            "Recall": round(recall[i], 3),
            "F1": round(f1[i], 3),
        })

results_df = pd.DataFrame(results_table)
print(results_df.to_string(index=False))
results_df.to_csv(os.path.join(CONFIG["output_dir"], "performance_table.csv"), index=False)

# %% CELL 11 — ROC curves + confusion matrix (Fig. 5), best model = MobileNetV2
def plot_roc_and_confusion(arch):
    y_true, y_pred, y_probs = predictions_store[arch]
    n_classes = len(CONFIG["classes"])
    y_true_bin = label_binarize(y_true, classes=range(n_classes))

    fig, axes = plt.subplots(1, 2, figsize=(13, 5))

    for i, cls in enumerate(CONFIG["classes"]):
        fpr, tpr, _ = roc_curve(y_true_bin[:, i], y_probs[:, i])
        roc_auc = auc(fpr, tpr)
        axes[0].plot(fpr, tpr, label=f"{cls} (AUC={roc_auc:.2f})")
    axes[0].plot([0, 1], [0, 1], "k--")
    axes[0].set_xlabel("False Positive Rate")
    axes[0].set_ylabel("True Positive Rate")
    axes[0].set_title(f"ROC — {arch}")
    axes[0].legend()

    cm = confusion_matrix(y_true, y_pred, normalize="true") * 100
    im = axes[1].imshow(cm, cmap="YlOrBr")
    axes[1].set_xticks(range(n_classes)); axes[1].set_xticklabels(CONFIG["classes"], rotation=45)
    axes[1].set_yticks(range(n_classes)); axes[1].set_yticklabels(CONFIG["classes"])
    axes[1].set_xlabel("Predicted Label"); axes[1].set_ylabel("True Label")
    axes[1].set_title(f"Confusion Matrix (%) — {arch}")
    for i in range(n_classes):
        for j in range(n_classes):
            axes[1].text(j, i, f"{cm[i, j]:.2f}", ha="center", va="center")
    fig.colorbar(im, ax=axes[1])
    plt.tight_layout()
    plt.savefig(os.path.join(CONFIG["output_dir"], f"roc_cm_{arch}.png"), dpi=150)
    plt.show()

for arch in architectures:
    plot_roc_and_confusion(arch)

# %% CELL 13 — Grad-CAM for the selected model (MobileNetV2)
def get_last_conv_layer_name(model):
    for layer in reversed(model.layers):
        if isinstance(layer, layers.Conv2D):
            return layer.name
    raise ValueError("No Conv2D layer found.")

def make_gradcam_heatmap(img_array, model, last_conv_layer_name, pred_index=None):
    grad_model = tf.keras.models.Model(
        [model.inputs], [model.get_layer(last_conv_layer_name).output, model.output]
    )
    with tf.GradientTape() as tape:
        conv_outputs, predictions = grad_model(img_array)
        if pred_index is None:
            pred_index = tf.argmax(predictions[0])
        class_channel = predictions[:, pred_index]

    grads = tape.gradient(class_channel, conv_outputs)
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    conv_outputs = conv_outputs[0]
    heatmap = conv_outputs @ pooled_grads[..., tf.newaxis]
    heatmap = tf.squeeze(heatmap)
    heatmap = tf.maximum(heatmap, 0) / (tf.math.reduce_max(heatmap) + 1e-8)
    return heatmap.numpy()

def overlay_gradcam(img_path, heatmap, alpha=0.4):
    img = cv2.imread(img_path)
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img = cv2.resize(img, (IMG_SIZE, IMG_SIZE))
    heatmap = cv2.resize(heatmap, (IMG_SIZE, IMG_SIZE))
    heatmap = np.uint8(255 * heatmap)
    heatmap = cv2.applyColorMap(heatmap, cv2.COLORMAP_JET)
    heatmap = cv2.cvtColor(heatmap, cv2.COLOR_BGR2RGB)
    overlay = np.uint8(img * (1 - alpha) + heatmap * alpha)
    return overlay

last_conv = get_last_conv_layer_name(best_model)
print("Using conv layer for Grad-CAM:", last_conv)

fig, axes = plt.subplots(1, len(CONFIG["classes"]), figsize=(15, 5))
for i, cls in enumerate(CONFIG["classes"]):
    row = test_df[test_df["Plane"] == cls].iloc[0]
    img = cv2.imread(row["filepath"])
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img = cv2.resize(img, (IMG_SIZE, IMG_SIZE)).astype(np.float32)
    img_norm = (img - eval_datagen.mean) / (eval_datagen.std + 1e-7)
    img_array = np.expand_dims(img_norm, axis=0)

    heatmap = make_gradcam_heatmap(img_array, best_model, last_conv)
    overlay = overlay_gradcam(row["filepath"], heatmap)

    axes[i].imshow(overlay)
    axes[i].set_title(cls)
    axes[i].axis("off")

plt.tight_layout()
plt.savefig(os.path.join(CONFIG["output_dir"], "gradcam_examples.png"), dpi=150)
