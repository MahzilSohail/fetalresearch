# FEENet-Fetal v5 — CLASS-BALANCED + STABILIZED EMA (6-class)
#
# WHAT CHANGED vs v4 (and WHY — each change targets a real, visible issue in
# your own training logs, not a guess at what might raise test accuracy):
#
#   1. EMA WARMUP (bug fix): v4's fixed high decay (0.999-0.9995) produced
#      near-random EMA weights in epoch 1 (your logs show EMA val acc of
#      20-43% while raw was 88-89% for all three models). Standard fix (as
#      used in torchvision/timm EMA implementations): decay ramps from 0 up
#      to the target value over the first ~2000 steps, so EMA never averages
#      in a near-random head.
#   2. EFFECTIVE NUMBER OF SAMPLES REWEIGHTING (Cui et al., CVPR 2019)
#      replaces raw inverse-frequency sampling. Raw inverse frequency gives
#      'abdomen' (317 train imgs) a >7x oversampling ratio vs 'other' (2328),
#      risking memorization of a small set. Effective-number weights soften
#      this in a principled, published way.
#   3. CLASS-BALANCED FOCAL LOSS (config-gated per model) replaces plain
#      CE+label-smoothing, so the loss itself — not just the sampler —
#      accounts for residual imbalance and down-weights easy majority-class
#      examples.
#   4. EARLY STOPPING in ensemble calibration: v4 trained a fixed 25 epochs
#      on a 768-image val set; your own log shows ValLoss plateaus by
#      epoch ~4-6, so the remaining epochs are pure overfit risk on a tiny
#      set. Added patience-based early stopping + L2 penalty on the mixing
#      weights.
#   5. EXPANDED TTA (10 passes instead of 8): one more crop-scale pass and
#      one flip+color-jitter combo, for extra inference-time diversity at
#      zero retraining cost.
#   6. OPTIONAL CLAHE local-contrast preprocessing — standard in ultrasound
#      image analysis literature, applied identically to train/val/test so
#      there is no leakage. Off by default (set APPLY_CLAHE=True to enable);
#      recrops from scratch if toggled since it changes cached PROC_DIR files.
#
# NOT added: SAM/ASAM, gradual unfreezing, a second stacking layer. Each would
# roughly double training compute or add another learned layer on the same
# 768-image val set, and none address a problem actually visible in your
# logs — complexity without a diagnosed cause is not something a reviewer
# rewards, and it's not something I'll claim helps just because it's on the
# "try these" list.
#
# HONEST EXPECTATION: these are legitimate fixes for real, log-visible issues
# (broken early EMA, imbalance handling, calibration overfitting) and should
# move accuracy/F1/calibration in the right direction. I'm not claiming they
# reach any specific target — only running this on your data will tell you.
#

_clahe = cv2.createCLAHE(clipLimit=CLAHE_CLIP_LIMIT, tileGridSize=CLAHE_TILE_GRID)

def _maybe_clahe(pil_img):
    """Local-contrast enhancement, applied identically to every image in every
    split (train/val/test) at preprocessing time -- this is a per-image
    deterministic transform, not a statistic learned from the data, so it
    introduces no leakage. Standard in ultrasound image analysis to counter
    speckle-related low local contrast."""
    if not APPLY_CLAHE:
        return pil_img
    arr = np.array(pil_img)
    arr = _clahe.apply(arr)
    return Image.fromarray(arr)

def crop_and_resize_all():
    n = 0
    for c in CLASSES:
        src_d = os.path.join(ORG_DIR, c); dst_d = os.path.join(PROC_DIR, c)
        os.makedirs(dst_d, exist_ok=True)
        if not os.path.exists(src_d): continue
        for f in os.listdir(src_d):
            if not f.lower().endswith(('.png', '.jpg', '.jpeg')): continue
            img = Image.open(os.path.join(src_d, f)).convert('L')
            w, h = img.size
            img = img.crop((35, 30, w - 35, h - 30)).resize(
                (224, 224), Image.Resampling.LANCZOS)
            img = _maybe_clahe(img)
            img.save(os.path.join(dst_d, f)); n += 1
    print(f"Cropped & resized {n} images -> {PROC_DIR} (CLAHE={'on' if APPLY_CLAHE else 'off'})")

class FetalUltrasoundDataset(Dataset):
    def __init__(self, root_dir, transform=None):
        self.samples, self.labels = [], []
        self.transform = transform
        for i, c in enumerate(CLASSES):
            d = os.path.join(root_dir, c)
            if not os.path.exists(d): continue
            for f in sorted(os.listdir(d)):
                if f.lower().endswith(('.png', '.jpg', '.jpeg')):
                    self.samples.append(os.path.join(d, f))
                    self.labels.append(i)
        print(f"Loaded {len(self.samples)} images across {NUM_CLASSES} classes")
    def __len__(self): return len(self.samples)
    def __getitem__(self, idx):
        img = Image.open(self.samples[idx]).convert('L')
        if self.transform: img = self.transform(img)
        return img, self.labels[idx]

class SubsetDataset(Dataset):
    def __init__(self, full_ds, indices, transform):
        self.full_ds, self.indices, self.transform = full_ds, indices, transform
    def __len__(self): return len(self.indices)
    def __getitem__(self, idx):
        img, label = self.full_ds[self.indices[idx]]
        if self.transform: img = self.transform(img)
        return img, label

def make_patient_disjoint_splits(samples, labels, csv_path, seed=SEED):
    df = pd.read_csv(csv_path, sep=';')
    img_to_patient = dict(zip(df['Image_name'], df['Patient_num']))
    has_train_col  = 'Train ' in df.columns or 'Train' in df.columns
    train_col      = 'Train ' if 'Train ' in df.columns else ('Train' if 'Train' in df.columns else None)
    img_to_train   = dict(zip(df['Image_name'], df[train_col])) if has_train_col else None

    indices = np.arange(len(samples))
    sample_patients = np.array([img_to_patient[os.path.splitext(os.path.basename(p))[0]] for p in samples])
    labels_arr = np.array(labels)

    if has_train_col:
        is_train = np.array([img_to_train[os.path.splitext(os.path.basename(p))[0]] == 1 for p in samples])
        test_idx       = indices[~is_train]
        train_full_idx = indices[is_train]
        gss = GroupShuffleSplit(n_splits=1, test_size=0.125, random_state=seed)
        train_rel, val_rel = next(gss.split(train_full_idx, labels_arr[train_full_idx], groups=sample_patients[train_full_idx]))
        train_idx = train_full_idx[train_rel]
        val_idx   = train_full_idx[val_rel]
        method = "burgos-artizzu_2020_official_train_test + patient-disjoint val carve-out"
    else:
        gss1 = GroupShuffleSplit(n_splits=1, test_size=0.30, random_state=seed)
        train_idx, temp_idx = next(gss1.split(indices, labels_arr, groups=sample_patients))
        gss2 = GroupShuffleSplit(n_splits=1, test_size=0.5, random_state=seed)
        val_rel, test_rel = next(gss2.split(temp_idx, labels_arr[temp_idx], groups=sample_patients[temp_idx]))
        val_idx  = temp_idx[val_rel]
        test_idx = temp_idx[test_rel]
        method = "patient_level_70_15_15_seed42"

    return train_idx, val_idx, test_idx, method, sample_patients

def effective_number_weights(counts, beta=EFFECTIVE_NUM_BETA):
    """Cui et al., CVPR 2019 'Class-Balanced Loss Based on Effective Number
    of Samples'. Softer than raw inverse-frequency: as beta -> 1 this
    approaches inverse-frequency, but for realistic beta (0.999-0.9999) it
    avoids the extreme oversampling ratios raw inverse-frequency produces
    (here: 7.3x for 'abdomen' vs 'other' under raw 1/n weighting)."""
    counts = np.maximum(counts, 1)
    effective_num = 1.0 - np.power(beta, counts)
    weights = (1.0 - beta) / effective_num
    weights = weights / weights.sum() * len(counts)   # normalize to mean 1
    return weights

class ClassBalancedFocalLoss(nn.Module):
    """Focal loss (Lin et al., 2017) with per-class weights from the effective-
    number-of-samples scheme (Cui et al., 2019), plus light label smoothing.
    Falls back to plain CE+label-smoothing if USE_FOCAL_LOSS=False, so this
    is a drop-in replacement for the original criterion.
    """
    def __init__(self, class_weights, gamma=FOCAL_GAMMA, label_smoothing=LABEL_SMOOTH,
                 use_focal=USE_FOCAL_LOSS):
        super().__init__()
        self.gamma = gamma
        self.ls = label_smoothing
        self.use_focal = use_focal
        self.register_buffer('weight', torch.tensor(class_weights, dtype=torch.float32))

    def forward(self, logits, target):
        n_classes = logits.size(1)
        w = self.weight.to(logits.device)
        if not self.use_focal:
            return F.cross_entropy(logits, target, weight=w, label_smoothing=self.ls)

        log_p = F.log_softmax(logits, dim=1)
        with torch.no_grad():
            true_dist = torch.full_like(log_p, self.ls / (n_classes - 1))
            true_dist.scatter_(1, target.unsqueeze(1), 1.0 - self.ls)
        p = log_p.exp()
        pt = (p * true_dist).sum(dim=1).clamp(min=1e-8)   # smoothed "prob of true class"
        focal_term = (1 - pt).pow(self.gamma)
        ce = -(true_dist * log_p).sum(dim=1)
        per_class_w = w[target]
        loss = per_class_w * focal_term * ce
        return loss.mean()

# ----------------------------- 9. MIXUP -------------------------------------
def mixup_data(x, y, alpha=MIXUP_ALPHA):
    lam   = np.random.beta(alpha, alpha) if alpha > 0 else 1.0
    index = torch.randperm(x.size(0), device=x.device)
    return lam * x + (1 - lam) * x[index], y, y[index], lam

def mixup_criterion(criterion, pred, y_a, y_b, lam):
    return lam * criterion(pred, y_a) + (1 - lam) * criterion(pred, y_b)

class ChannelAttention(nn.Module):
    def __init__(self, channels, reduction=16):
        super().__init__()
        self.avg_pool = nn.AdaptiveAvgPool2d(1)
        self.max_pool = nn.AdaptiveMaxPool2d(1)
        self.fc = nn.Sequential(nn.Linear(channels, channels // reduction, bias=False), nn.ReLU(), nn.Linear(channels // reduction, channels, bias=False))
        self.sigmoid = nn.Sigmoid()
    def forward(self, x):
        b, c, _, _ = x.shape
        a = self.fc(self.avg_pool(x).view(b, c))
        m = self.fc(self.max_pool(x).view(b, c))
        return x * self.sigmoid((a + m).view(b, c, 1, 1))

class SpatialAttention(nn.Module):
    def __init__(self, kernel_size=7):
        super().__init__()
        self.conv = nn.Conv2d(2, 1, kernel_size, padding=kernel_size//2, bias=False)
        self.sigmoid = nn.Sigmoid()
    def forward(self, x):
        avg = torch.mean(x, dim=1, keepdim=True)
        mx, _ = torch.max(x, dim=1, keepdim=True)
        return x * self.sigmoid(self.conv(torch.cat([avg, mx], dim=1)))

class CBAM(nn.Module):
    def __init__(self, channels, reduction=16):
        super().__init__()
        self.ca = ChannelAttention(channels, reduction)
        self.sa = SpatialAttention()
    def forward(self, x): return self.sa(self.ca(x))

class MultiScaleFusion(nn.Module):
    def __init__(self, in_channels_list, out_channels=256):
        super().__init__()
        self.lateral = nn.ModuleList([nn.Conv2d(c, out_channels, 1, bias=False) for c in in_channels_list])
        self.fuse = nn.Sequential(nn.Conv2d(out_channels * len(in_channels_list), out_channels, 1, bias=False), nn.BatchNorm2d(out_channels), nn.ReLU(inplace=True))
        self.pool = nn.AdaptiveAvgPool2d(1)
    def forward(self, features):
        target = features[0].shape[-2:]
        lats = [F.interpolate(L(f), size=target, mode='nearest') for L, f in zip(self.lateral, features)]
        return self.pool(self.fuse(torch.cat(lats, dim=1))).flatten(1)

class SpeckleReductionBlock(nn.Module):
    def __init__(self, channels):
        super().__init__()
        self.dw = nn.Conv2d(channels, channels, 3, padding=1, groups=channels, bias=False)
        self.bn = nn.BatchNorm2d(channels)
        self.relu = nn.ReLU(inplace=True)
    def forward(self, x): return x + self.relu(self.bn(self.dw(x)))

class TransformerEncoderBlock(nn.Module):
    def __init__(self, embed_dim, num_heads=4, mlp_ratio=2.0, dropout=0.1):
        super().__init__()
        self.norm1 = nn.LayerNorm(embed_dim)
        self.attn = nn.MultiheadAttention(embed_dim, num_heads, dropout=dropout, batch_first=True)
        self.norm2 = nn.LayerNorm(embed_dim)
        h = int(embed_dim * mlp_ratio)
        self.mlp = nn.Sequential(nn.Linear(embed_dim, h), nn.GELU(), nn.Dropout(dropout), nn.Linear(h, embed_dim), nn.Dropout(dropout))
    def forward(self, x):
        z = self.norm1(x)
        x = x + self.attn(z, z, z)[0]
        return x + self.mlp(self.norm2(x))

def _init_head_weights(m):
    """Sane init for freshly-added heads (helps B5 transformer stability)."""
    if isinstance(m, nn.Linear):
        nn.init.xavier_uniform_(m.weight)
        if m.bias is not None: nn.init.zeros_(m.bias)
    elif isinstance(m, nn.LayerNorm):
        nn.init.ones_(m.weight); nn.init.zeros_(m.bias)

class AttnEfficientNetB0(nn.Module):
    def __init__(self, num_classes=NUM_CLASSES, pretrained=True, dropout=DROPOUT):
        super().__init__()
        self.backbone = timm.create_model('efficientnet_b0', pretrained=pretrained, num_classes=0, global_pool='')
        d = self.backbone.num_features
        self.cbam = CBAM(d)
        self.pool = nn.AdaptiveAvgPool2d(1)
        self.feat_dim = d
        self.classifier = nn.Sequential(nn.Dropout(dropout), nn.Linear(d, num_classes))
    def forward_features(self, x): return self.pool(self.cbam(self.backbone(x))).flatten(1)
    def forward(self, x): return self.classifier(self.forward_features(x))
    @property
    def gradcam_target_layer(self): return self.backbone.conv_head

class MSEfficientNetB4NS(nn.Module):
    def __init__(self, num_classes=NUM_CLASSES, pretrained=True, out_channels=256, dropout=DROPOUT):
        super().__init__()
        self.backbone = timm.create_model('tf_efficientnet_b4_ns', pretrained=pretrained, features_only=True, out_indices=(2,3,4))
        in_ch = self.backbone.feature_info.channels()
        self.fusion = MultiScaleFusion(in_ch, out_channels)
        self.feat_dim = out_channels
        self.classifier = nn.Sequential(nn.Dropout(dropout), nn.Linear(out_channels, num_classes))
    def forward_features(self, x): return self.fusion(self.backbone(x))
    def forward(self, x): return self.classifier(self.forward_features(x))
    @property
    def gradcam_target_layer(self): return self.fusion.fuse[0]

class USEfficientNetB5NS(nn.Module):
    def __init__(self, num_classes=NUM_CLASSES, pretrained=True, num_heads=4, dropout=DROPOUT):
        super().__init__()
        self.backbone = timm.create_model('tf_efficientnet_b5_ns', pretrained=pretrained, num_classes=0, global_pool='')
        d = self.backbone.num_features
        self.speckle = SpeckleReductionBlock(d)
        self.pool_tokens = nn.AdaptiveAvgPool2d((7, 7))
        self.transformer = TransformerEncoderBlock(d, num_heads=num_heads)
        self.norm = nn.LayerNorm(d)
        self.feat_dim = d
        self.classifier = nn.Sequential(nn.Dropout(dropout), nn.Linear(d, num_classes))
        # init the freshly-added head modules (NOT the pretrained backbone)
        self.transformer.apply(_init_head_weights)
        self.norm.apply(_init_head_weights)
        self.classifier.apply(_init_head_weights)
    def forward_features(self, x):
        f = self.speckle(self.backbone(x))
        f = self.pool_tokens(f)
        tokens = self.transformer(f.flatten(2).transpose(1, 2))
        return self.norm(tokens.mean(dim=1))
    def forward(self, x): return self.classifier(self.forward_features(x))
    @property
    def gradcam_target_layer(self): return self.backbone.conv_head

# --- Ensemble (unchanged from your v3 fix: prob-space mix + learned alpha/temp)
class FEENet(nn.Module):
    def __init__(self, models_list):
        super().__init__()
        self.models = nn.ModuleList(models_list)
        n = len(models_list)
        self.alpha = nn.Parameter(torch.ones(n))
        self.log_temp = nn.Parameter(torch.zeros(n))
        for m in self.models:
            for p in m.parameters():
                p.requires_grad = False
    def forward(self, x):
        w = F.softmax(self.alpha, dim=0)
        probs = []
        for i, m in enumerate(self.models):
            logits = m(x)
            T = torch.exp(self.log_temp[i]).clamp(min=0.05, max=5.0)
            probs.append(F.softmax(logits / T, dim=1))
        probs = torch.stack(probs, dim=0)
        mixed = (probs * w.view(-1, 1, 1)).sum(dim=0)
        return torch.log(mixed.clamp_min(1e-12))

# ----------------------------- 12. EMA ---------------------------------------
class ModelEMA:
    """Exponential moving average of weights. Evaluated alongside the raw model;
    the better of the two (by macro-F1) is kept each epoch.

    v5 fix: decay ramps from 0 up to the target decay over EMA_WARMUP_STEPS
    optimizer steps (same trick used in torchvision/timm EMA utilities).
    Without this, a fixed high decay (e.g. 0.999) means the EMA is still
    ~37% original-random-head-weight after 1000 steps, which is exactly
    what the v4 logs showed (epoch-1 EMA val acc of 20-43% vs 88-89% raw).
    """
    def __init__(self, model, decay=0.999, warmup_steps=EMA_WARMUP_STEPS):
        self.ema = copy.deepcopy(model).eval()
        for p in self.ema.parameters(): p.requires_grad_(False)
        self.target_decay = decay
        self.warmup_steps = warmup_steps
        self.step_count = 0
    def _current_decay(self):
        # Standard EMA warmup ramp (used in YOLOv5/timm-style EMA utilities):
        # (1+t)/(10+t) rises quickly from ~0.1 and asymptotes to 1.0, so the
        # min() with target_decay means early steps use a LOW decay (EMA
        # tracks the raw model closely while it's still near-random) and
        # later steps use the intended target_decay for smoothing.
        ramped = (1 + self.step_count) / (10 + self.step_count)
        return min(self.target_decay, ramped)
    @torch.no_grad()
    def update(self, model):
        self.step_count += 1
        d = self._current_decay()
        for e, m in zip(self.ema.state_dict().values(), model.state_dict().values()):
            if e.dtype.is_floating_point:
                e.mul_(d).add_(m.detach().to(e.dtype), alpha=1 - d)
            else:
                e.copy_(m)

# ----------------------------- 13. TRAIN UTILS -------------------------------
# NOTE: eval-time loss uses UNWEIGHTED CE (no class weights, no focal term) --
# we want val/test loss to reflect the true data distribution for honest
# monitoring and for the ensemble-calibration NLL below. Only the TRAINING
# criterion uses class-balanced focal loss.
CE_EVAL = nn.CrossEntropyLoss(label_smoothing=LABEL_SMOOTH)
criterion = CE_EVAL  # module-level name kept for compatibility
TRAIN_CRITERION = ClassBalancedFocalLoss(CLASS_WEIGHTS)

def build_param_groups(model, backbone_lr, head_lr, weight_decay):
    """Discriminative LR: pretrained backbone vs freshly-initialized head."""
    backbone, head = [], []
    for n, p in model.named_parameters():
        if not p.requires_grad: continue
        (backbone if n.startswith('backbone.') else head).append(p)
    groups = []
    if backbone: groups.append({'params': backbone, 'lr': backbone_lr, 'weight_decay': weight_decay})
    if head:     groups.append({'params': head,     'lr': head_lr,     'weight_decay': weight_decay})
    return groups, [backbone_lr, head_lr][:len(groups)]

def train_one_epoch_cfg(model, loader, criterion, optimizer, scheduler, ema, cfg):
    model.train()
    tot_loss, correct, total = 0.0, 0, 0
    ma, clip = cfg['mixup_alpha'], cfg['grad_clip']
    for x, y in loader:
        x = x.to(device, non_blocking=True); y = y.to(device, non_blocking=True)
        optimizer.zero_grad()
        if ma > 0 and random.random() < 0.5:
            x_mix, y_a, y_b, lam = mixup_data(x, y, ma)
            out = model(x_mix)
            loss = mixup_criterion(criterion, out, y_a, y_b, lam)
            correct += (out.argmax(1) == y_a).sum().item()
        else:
            out = model(x)
            loss = criterion(out, y)
            correct += (out.argmax(1) == y).sum().item()
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=clip)
        optimizer.step(); scheduler.step()
        ema.update(model)
        tot_loss += loss.item() * x.size(0)
        total += y.size(0)
    return tot_loss / total, correct / total * 100

@torch.no_grad()
def evaluate(model, loader, criterion=None):
    model.eval()
    preds_all, trues_all, probs_all = [], [], []
    tot_loss, total = 0.0, 0
    for x, y in loader:
        x = x.to(device, non_blocking=True)
        y_dev = y.to(device, non_blocking=True)
        out = model(x)
        if criterion is not None:
            tot_loss += criterion(out, y_dev).item() * x.size(0)
        preds_all.extend(out.argmax(1).cpu().numpy())
        trues_all.extend(y.numpy())
        probs_all.append(F.softmax(out, dim=1).cpu().numpy())
        total += y.size(0)
    preds = np.array(preds_all); trues = np.array(trues_all)
    probs = np.vstack(probs_all)
    acc = (preds == trues).mean() * 100
    loss = (tot_loss / total) if criterion is not None else float('nan')
    return loss, acc, preds, trues, probs

def val_scores(model):
    loss, acc, preds, trues, _ = evaluate(model, val_loader, criterion=CE_EVAL)
    f1 = f1_score(trues, preds, average='macro', zero_division=0) * 100
    return loss, acc, f1

@torch.no_grad()
def evaluate_tta(model, full_ds_raw, test_indices):
    model.eval()
    all_probs = None
    all_trues = np.array([full_ds_raw.labels[i] for i in test_indices])
    for tf in tta_transforms[:TTA_PASSES]:
        tta_ds = SubsetDataset(full_ds_raw, test_indices, tf)
        tta_loader = DataLoader(tta_ds, batch_size=BATCH_SIZE, shuffle=False, num_workers=2, pin_memory=True)
        probs_this = []
        for x, _ in tta_loader:
            out = model(x.to(device))
            probs_this.append(F.softmax(out, dim=1).cpu().numpy())
        probs_this = np.vstack(probs_this)
        all_probs = probs_this if all_probs is None else all_probs + probs_this
    all_probs /= TTA_PASSES
    preds = all_probs.argmax(axis=1)
    acc = (preds == all_trues).mean() * 100
    return acc, preds, all_trues, all_probs

# --- post-hoc ensemble calibration (v5: early stopping + L2 on alpha) -------
def train_ensemble_weights(feenet, val_loader, epochs=ENSEMBLE_CALIB_EPOCHS, lr=ENSEMBLE_CALIB_LR,
                            patience=ENSEMBLE_CALIB_PATIENCE, l2_alpha=ENSEMBLE_CALIB_L2):
    """v5 change: the original trained a fixed 25 epochs on a 768-image val
    set; its own log shows ValLoss plateauing by epoch ~4-6, meaning the rest
    of training only risks overfitting the calibration weights to this small
    set. We now (a) track the best-val-loss state and restore it, (b) stop
    early after `patience` epochs without improvement, (c) add a small L2
    penalty on alpha so the learned mixing weights don't drift to extreme
    values chasing val-set noise.
    """
    for m in feenet.models: m.eval()
    optimizer = optim.Adam([feenet.alpha, feenet.log_temp], lr=lr)
    criterion = nn.NLLLoss()
    print(f"\n{'='*60}\nCalibrating FEENet ensemble weights on val set (early stopping, patience={patience})\n{'='*60}")

    best_loss, best_state, epochs_no_improve = float('inf'), None, 0
    for ep in range(1, epochs + 1):
        tot_loss, correct, total = 0.0, 0, 0
        for x, y in val_loader:
            x, y = x.to(device, non_blocking=True), y.to(device, non_blocking=True)
            optimizer.zero_grad()
            log_mixed = feenet(x)
            loss = criterion(log_mixed, y) + l2_alpha * feenet.alpha.pow(2).sum()
            loss.backward(); optimizer.step()
            tot_loss += loss.item() * x.size(0)
            correct += (log_mixed.argmax(1) == y).sum().item()
            total += y.size(0)
        ep_loss = tot_loss / total
        with torch.no_grad():
            w = F.softmax(feenet.alpha, dim=0).cpu().numpy()
            temps = torch.exp(feenet.log_temp).cpu().numpy()
        print(f"[EnsembleCalib] Ep {ep:2d} | ValLoss {ep_loss:.4f} | "
              f"ValAcc {correct/total*100:.2f}% | weights={np.round(w, 3)} | temps={np.round(temps, 3)}")

        if ep_loss < best_loss - 1e-5:
            best_loss = ep_loss
            best_state = {'alpha': feenet.alpha.detach().clone(),
                          'log_temp': feenet.log_temp.detach().clone()}
            epochs_no_improve = 0
        else:
            epochs_no_improve += 1
            if epochs_no_improve >= patience:
                print(f"[EnsembleCalib] Early stopping at epoch {ep} "
                      f"(no improvement for {patience} epochs). Restoring best state (ValLoss {best_loss:.4f}).")
                break

    if best_state is not None:
        with torch.no_grad():
            feenet.alpha.copy_(best_state['alpha'])
            feenet.log_temp.copy_(best_state['log_temp'])
    return feenet

TRAINING_HISTORY = {}

def train_submodel_tuned(model, name, cfg):
    print(f"\n{'='*60}\nTraining {name}  (per-model tuned)\n{'='*60}")
    print(f"  epochs={cfg['epochs']} backbone_lr={cfg['backbone_lr']} head_lr={cfg['head_lr']} "
          f"wd={cfg['weight_decay']} warmup={cfg['warmup_pct']} clip={cfg['grad_clip']} aug={cfg['aug']}")
    model = model.to(device)
    loader = LOADER_FOR[cfg['aug']]

    groups, max_lrs = build_param_groups(model, cfg['backbone_lr'], cfg['head_lr'], cfg['weight_decay'])
    optimizer = optim.AdamW(groups)
    scheduler = optim.lr_scheduler.OneCycleLR(
        optimizer, max_lr=max_lrs, epochs=cfg['epochs'], steps_per_epoch=len(loader),
        pct_start=cfg['warmup_pct'], div_factor=25.0, final_div_factor=1e3)
    train_criterion = TRAIN_CRITERION
    ema = ModelEMA(model, decay=cfg['ema_decay'])

    hist = {'train_acc': [], 'val_acc': [], 'train_loss': [], 'val_loss': []}
    best_f1, best_acc, best_state = -1.0, 0.0, None

    for ep in range(1, cfg['epochs'] + 1):
        tl, ta = train_one_epoch_cfg(model, loader, train_criterion, optimizer, scheduler, ema, cfg)

        vl_r, va_r, vf_r = val_scores(model)       # raw weights
        vl_e, va_e, vf_e = val_scores(ema.ema)     # EMA weights

        # pick the better view this epoch by macro-F1
        if vf_e >= vf_r:
            cand_state, cand_f1, cand_acc, cand_vl, cand_va, tag = ema.ema.state_dict(), vf_e, va_e, vl_e, va_e, 'ema'
        else:
            cand_state, cand_f1, cand_acc, cand_vl, cand_va, tag = model.state_dict(), vf_r, va_r, vl_r, va_r, 'raw'

        hist['train_acc'].append(ta); hist['val_acc'].append(cand_va)
        hist['train_loss'].append(tl); hist['val_loss'].append(cand_vl)

        if cand_f1 > best_f1:
            best_f1, best_acc = cand_f1, cand_acc
            best_state = copy.deepcopy(cand_state)

        print(f"[{name}] Ep {ep:2d} | Train {ta:5.2f}% | "
              f"Val(raw) acc {va_r:5.2f}% f1 {vf_r:5.2f} | "
              f"Val(ema) acc {va_e:5.2f}% f1 {vf_e:5.2f} | "
              f"kept={tag} | BestF1 {best_f1:5.2f} (acc {best_acc:5.2f}%)")

    model.load_state_dict(best_state); model.eval()
    torch.save(best_state, os.path.join(MODELS_DIR, f"{name}.pth"))
    TRAINING_HISTORY[name] = hist
    return model

def plot_roc_curves(probs, trues):
    y_bin = label_binarize(trues, classes=range(NUM_CLASSES))
    fpr_macro = np.linspace(0, 1, 2000); tpr_macro = np.zeros_like(fpr_macro)
    colors = ['blue', 'green', 'orange', 'red', 'purple', 'brown']
    fig, axes = plt.subplots(1, 2, figsize=(18, 8))
    for i in range(NUM_CLASSES):
        fpr, tpr, _ = roc_curve(y_bin[:, i], probs[:, i]); roc_auc = auc(fpr, tpr)
        tpr_macro += np.interp(fpr_macro, fpr, tpr)
        axes[0].plot(fpr, tpr, color=colors[i], lw=2.2, label=f'{CLASSES[i]} (AUC={roc_auc:.5f})')
        axes[1].plot(fpr, tpr, color=colors[i], lw=2.2, label=f'{CLASSES[i]} (AUC={roc_auc:.5f})')
    tpr_macro /= NUM_CLASSES; macro_auc = auc(fpr_macro, tpr_macro)
    for ax in axes:
        ax.plot(fpr_macro, tpr_macro, 'navy', linestyle='--', lw=3, label=f'Macro AUC = {macro_auc:.5f}')
        ax.plot([0,1],[0,1], 'k--', lw=1)
        ax.set_xlabel('False Positive Rate'); ax.set_ylabel('True Positive Rate')
        ax.legend(loc='lower right'); ax.grid(alpha=0.3)
    axes[0].set_title('(a) Linear Scale'); axes[1].set_title('(b) Log Scale')
    axes[1].set_xscale('log'); axes[1].set_xlim([1e-4, 1])
    fig.suptitle('FEENet ROC Curves', fontsize=15, fontweight='bold')
    plt.tight_layout()
    save_path = os.path.join(PLOTS_DIR, "fig_06_roc_curves.png")
    fig.savefig(save_path, dpi=220, bbox_inches='tight'); plt.show()
    print(f"ROC Plot saved & visible: {save_path}")
    return macro_auc
